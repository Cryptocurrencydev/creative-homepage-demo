Usage

import "./kit/style/kit.css";
import "./kit/motion/motion.css";
import { createCursor, createTransition, reveal,
         scrubWords, magnetic, squish } from "./kit/motion/index.js";

createCursor();                       // auto-off on touch
const t = createTransition({ text: "wobble…" });
t.bindLinks('a[href^="#"][data-transition]');
reveal("[data-reveal]", { split: "words" });
scrubWords(document.querySelector(".manifesto"));
document.querySelectorAll(".k-btn").forEach((b) => { magnetic(b); squish(b); });
